cmake_minimum_required(VERSION 3.22.1)
project(clipforge_whisper)
include(FetchContent)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
FetchContent_Declare(whisper_cpp GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG v1.7.6 GIT_SHALLOW TRUE GIT_SUBMODULES ggml GIT_SUBMODULES_RECURSE TRUE)
FetchContent_MakeAvailable(whisper_cpp)
add_library(clipforge_whisper SHARED whisper_jni.cpp)
find_library(log-lib log)
target_link_libraries(clipforge_whisper PRIVATE whisper ${log-lib})
